wnba_leagueplayerondetails(

  game_segment = "",
  last_n_games = 0,
  league_id = "10",
  location = "",
  measure_type = "Base",
  month = 0,
  opponent_team_id = 0,
  outcome = "",
  pace_adjust = "N",
  per_mode = "Totals",
  period = 0,

  season_segment = "",
  season_type = "Regular Season",

  vs_conference = "",
  vs_division = "",
  ...
)
